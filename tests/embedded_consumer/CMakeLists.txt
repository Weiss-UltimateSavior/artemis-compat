cmake_minimum_required(VERSION 3.16)
project(artemis_embedded_consumer LANGUAGES C CXX)
enable_testing()
set(ARTC_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Artemis source checkout")
add_subdirectory("${ARTC_SOURCE_DIR}" artemis)
foreach(unwanted artc artemis artemis-mac)
    if(TARGET ${unwanted})
        message(FATAL_ERROR "Embedding unexpectedly created host target ${unwanted}")
    endif()
endforeach()
add_library(embedded_probe SHARED probe.cpp)
target_link_libraries(embedded_probe PRIVATE artemis::core)
if(ANDROID OR OHOS OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(embedded_probe PRIVATE "LINKER:--no-undefined")
endif()
add_executable(embedded_consumer main.cpp)
target_link_libraries(embedded_consumer PRIVATE embedded_probe)
add_test(NAME embedded_consumer COMMAND embedded_consumer)
